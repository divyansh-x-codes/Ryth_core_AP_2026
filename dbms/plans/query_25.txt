--- [SQL STATEMENT] ---
SELECT
    s.id,
    s.title,
    s.excerpt,
    s."coverImage",
    b."createdAt"
FROM "Bookmark" b
JOIN story s ON b."storyId" = s.id
WHERE b."userId" = 'USER_ID'
ORDER BY b."createdAt" DESC;
