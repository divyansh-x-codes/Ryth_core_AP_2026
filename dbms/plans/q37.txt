--- [SQL STATEMENT] ---
SELECT
    u.name,
    u.email,
    p.name AS plan_name,
    sub."startDate",
    sub."endDate"
FROM "Subscription" sub
JOIN "user" u ON sub."userId" = u.id
JOIN "SubscriptionPlan" p ON sub."planId" = p.id
WHERE sub.status = 'ACTIVE'
AND sub."endDate" > CURRENT_TIMESTAMP;
