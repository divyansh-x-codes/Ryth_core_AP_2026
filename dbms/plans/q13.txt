--- [SQL STATEMENT] ---
SELECT
    s.id,
    s.title,
    s.excerpt,
    u.name AS author
FROM story s
JOIN "user" u ON s."authorId" = u.id
WHERE s."categoryId" = 1
AND s.visibility = 'PUBLIC'
ORDER BY s."publishedAt" DESC;
