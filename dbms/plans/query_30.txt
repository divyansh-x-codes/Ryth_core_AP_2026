--- [SQL STATEMENT] ---
SELECT
    s.id,
    s.title,
    s."coverImage",
    u.name AS author,
    COUNT(l.id) AS total_likes
FROM story s
JOIN "user" u ON s."authorId" = u.id
LEFT JOIN "Like" l ON s.id = l."storyId"
WHERE s.visibility = 'PUBLIC'
GROUP BY s.id, s.title, s."coverImage", u.name
ORDER BY total_likes DESC
LIMIT 10;
