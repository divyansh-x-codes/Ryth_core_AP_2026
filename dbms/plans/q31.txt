--- [SQL STATEMENT] ---
SELECT
    u.id,
    u.name,
    COUNT(s.id) AS published_stories
FROM "user" u
JOIN story s ON u.id = s."authorId"
WHERE s.visibility = 'PUBLIC'
GROUP BY u.id, u.name
ORDER BY published_stories DESC
LIMIT 10;
