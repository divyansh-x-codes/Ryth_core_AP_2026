--- [SQL STATEMENT] ---
SELECT
    c.id,
    c.content,
    c."createdAt",
    u.name AS commenter,
    u.image AS profile_image
FROM "Comment" c
JOIN "user" u ON c."userId" = u.id
WHERE c."storyId" = 1
ORDER BY c."createdAt" DESC;
