--- [SQL STATEMENT] ---
SELECT
    s.id,
    s.title,
    COUNT(l.id) AS total_likes
FROM story s
LEFT JOIN "Like" l ON s.id = l."storyId"
GROUP BY s.id, s.title
ORDER BY total_likes DESC;
