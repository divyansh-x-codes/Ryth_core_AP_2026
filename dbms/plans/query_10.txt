--- [SQL STATEMENT] ---
SELECT
    s.id,
    s.title,
    s.slug,
    s.excerpt,
    s."coverImage",
    s."publishedAt",
    u.name AS author,
    c.name AS category
FROM story s
JOIN "user" u ON s."authorId" = u.id
JOIN "Category" c ON s."categoryId" = c.id
WHERE s.visibility = 'PUBLIC'
ORDER BY s."publishedAt" DESC;
