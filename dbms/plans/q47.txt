--- [SQL STATEMENT] ---
SELECT id, title, "publishedAt"
FROM story
WHERE visibility = 'PUBLIC'
AND "publishedAt" >= DATE_TRUNC('month', CURRENT_TIMESTAMP)
ORDER BY "publishedAt" DESC;
