--- [SQL STATEMENT] ---
SELECT
    pay.id,
    pay.amount,
    pay.currency,
    pay.status,
    pay."paymentProvider",
    pay."providerPaymentId",
    pay."paymentDate",
    sub.id AS subscription_id
FROM "Payment" pay
JOIN "Subscription" sub ON pay."subscriptionId" = sub.id
WHERE sub."userId" = 'USER_ID'
ORDER BY pay."createdAt" DESC;
