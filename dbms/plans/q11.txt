--- [SQL STATEMENT] ---
SELECT
    s.*,
    u.name AS author,
    c.name AS category
FROM story s
JOIN "user" u ON s."authorId" = u.id
JOIN "Category" c ON s."categoryId" = c.id
WHERE s.slug = 'sample-story';
