--- [SQL STATEMENT] ---
SELECT
    sub.id,
    u.name AS user_name,
    p.name AS plan_name,
    sub.status,
    sub."startDate",
    sub."endDate"
FROM "Subscription" sub
JOIN "user" u ON sub."userId" = u.id
JOIN "SubscriptionPlan" p ON sub."planId" = p.id
WHERE sub."userId" = 'USER_ID'
ORDER BY sub."createdAt" DESC;
