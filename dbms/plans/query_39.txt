--- [SQL STATEMENT] ---
UPDATE "Subscription"
SET status = 'EXPIRED',
    "updatedAt" = CURRENT_TIMESTAMP
WHERE status = 'ACTIVE'
AND "endDate" <= CURRENT_TIMESTAMP;
