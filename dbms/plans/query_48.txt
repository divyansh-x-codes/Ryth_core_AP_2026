--- [SQL STATEMENT] ---
SELECT
    s.id,
    s.title,
    COUNT(c.id) AS total_comments
FROM story s
LEFT JOIN "Comment" c ON s.id = c."storyId"
WHERE s.visibility = 'PUBLIC'
GROUP BY s.id, s.title
ORDER BY total_comments DESC
LIMIT 10;
