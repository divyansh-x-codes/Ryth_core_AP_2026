--- [SQL STATEMENT] ---
SELECT
    c.id,
    c.name,
    COUNT(s.id) AS total_stories
FROM "Category" c
LEFT JOIN story s ON s."categoryId" = c.id
GROUP BY c.id, c.name
ORDER BY total_stories DESC;
